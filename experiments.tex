\begin{tikzpicture}[
    >=stealth,
    % Define styles for different node types
    process/.style={
        draw, 
        rounded corners, 
        minimum height=1cm, 
        align=center, 
        fill=white,
        inner sep=6pt
    },
    meeting/.style={
        process, 
        draw=green!60!black, 
        text=green!60!black, 
        pattern=north east lines, 
        pattern color=green!60!black!20
    },
    procurement/.style={
        process, 
        draw=red, 
        text=red, 
        pattern=north east lines, 
        pattern color=red!20
    },
    aux/.style={
        process, 
        pattern=north east lines, 
        pattern color=gray!20
    },
    label_text/.style={
        font=\scriptsize,
        color=black
    },
    green_label/.style={
        font=\scriptsize,
        color=green!60!black
    }
]

    % --- Nodes ---

    % Planning
    \node[meeting] (planning) {Planning};
    \node[green_label, below=0.1cm of planning] {Meeting};

    % Design
    \node[process, right=1cm of planning] (design) {Design};
    \node[label_text, above=0.2cm of design] {1-2 weeks};

    % Parallel Phase Group
    % We position Manufacturing first to establish the center line, then place others relative to it
    \node[process, right=2.5cm of design] (manuf) {Manufacturing (Internal)};
    
    \node[procurement, above=0.3cm of manuf] (procure) {Procurement (External)};
    \node[label_text, above=0.2cm of procure] {1 week};

    \node[process, below=0.3cm of manuf, text width=4cm] (software) {Software Development\\Software Verification};
    
    \node[aux, below=0.3cm of software, text width=4cm] (aux) {Auxiliary Activities};

    % Integration
    \node[process, right=1cm of manuf] (integration) {Integration};
    \node[label_text, above=0.2cm of integration] {1 week};

    % Verification
    \node[process, right=1cm of integration, text width=3.5cm] (verification) {Verification, Validation\\and Improvement};
    \node[label_text, above=0.2cm of verification] {1-2 weeks};

    % Reviews
    \node[meeting, right=0.8cm of verification] (review) {Sprint Review};
    
    \node[meeting, right=0.8cm of review] (retro) {Sprint Retrospective};
    \node[green_label, below=0.1cm of retro] {Meetings};

    % --- Connections ---

    % Linear Flow
    \draw[->] (planning) -- (design);

    % Diverging from Design
    \draw[->] (design.east) -- (procure.west);
    \draw[->] (design.east) -- (manuf.west);
    \draw[->] (design.east) -- (software.west);
    \draw[->] (design.east) -- (aux.west);

    % Converging to Integration
    \draw[->] (procure.east) -- (integration.north west);
    \draw[->] (manuf.east) -- (integration.west);

    % Converging to Verification
    \draw[->] (integration) -- (verification);
    % Software and Aux connect directly to Verification in the diagram
    \draw[->] (software.east) -- (verification.south west);
    \draw[->] (aux.east) -- (verification.south west);

    % End sequence
    \draw[->] (verification) -- (review);
    \draw[->] (review) -- (retro);

    % Feedback Loop
    \draw[->, color=green!60!black] (retro.east) to[out=-90,in=-90, looseness=0.4] (planning.south);

\end{tikzpicture}
